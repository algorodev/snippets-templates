Evaluates a password against a set of configurable rules and returns a numeric score alongside the list of requirements that were not satisfied, making it easy to display actionable feedback in a registration form.

## Usage

```typescript
export interface PasswordOptions {
  /** Minimum number of characters required. Default: 8 */
  minLength?: number
  /** Require at least one uppercase letter. Default: true */
  requireUppercase?: boolean
  /** Require at least one lowercase letter. Default: true */
  requireLowercase?: boolean
  /** Require at least one digit. Default: true */
  requireDigit?: boolean
  /** Require at least one special character. Default: true */
  requireSpecial?: boolean
  /** Custom set of special characters to accept. Default: !@#$%^&*()_+-=[]{}|;':",.<>?/`~ */
  specialChars?: string
}

export interface PasswordValidationResult {
  valid: boolean
  /** Number of rules the password satisfies (0–5 by default). */
  score: number
  /** Human-readable descriptions of unmet requirements. */
  unmet: string[]
}

const DEFAULT_SPECIAL = `!@#$%^&*()_+-=[]{}|;':",.<>?/\`~`

/**
 * Validates a password against configurable strength rules.
 *
 * @param password - The password string to evaluate.
 * @param options  - Optional rule overrides.
 * @returns Validation result with score and list of unmet requirements.
 */
function isStrongPassword(
  password: string,
  options: PasswordOptions = {}
): PasswordValidationResult {
  const {
    minLength = 8,
    requireUppercase = true,
    requireLowercase = true,
    requireDigit = true,
    requireSpecial = true,
    specialChars = DEFAULT_SPECIAL,
  } = options

  const rules: Array<{ check: boolean; message: string }> = []

  rules.push({
    check: typeof password === 'string' && password.length >= minLength,
    message: `At least ${minLength} characters`,
  })

  if (requireUppercase) {
    rules.push({
      check: /[A-Z]/.test(password),
      message: 'At least one uppercase letter (A–Z)',
    })
  }

  if (requireLowercase) {
    rules.push({
      check: /[a-z]/.test(password),
      message: 'At least one lowercase letter (a–z)',
    })
  }

  if (requireDigit) {
    rules.push({
      check: /\d/.test(password),
      message: 'At least one digit (0–9)',
    })
  }

  if (requireSpecial) {
    const escapedSpecial = specialChars.replace(/[-[\]{}()*+?.,\\^$|#\s]/g, '\\$&')
    const specialRegex = new RegExp(`[${escapedSpecial}]`)
    rules.push({
      check: specialRegex.test(password),
      message: `At least one special character (${specialChars})`,
    })
  }

  const passed = rules.filter((r) => r.check)
  const unmet = rules.filter((r) => !r.check).map((r) => r.message)

  return {
    valid: unmet.length === 0,
    score: passed.length,
    unmet,
  }
}

export default isStrongPassword
```

## Options

| Option | Type | Default | Description |
|---|---|---|---|
| `minLength` | `number` | `8` | Minimum character count |
| `requireUppercase` | `boolean` | `true` | Must contain A–Z |
| `requireLowercase` | `boolean` | `true` | Must contain a–z |
| `requireDigit` | `boolean` | `true` | Must contain 0–9 |
| `requireSpecial` | `boolean` | `true` | Must contain a special character |
| `specialChars` | `string` | see above | Override the accepted special character set |

## Examples

```typescript
isStrongPassword('hunter2')
// { valid: false, score: 2, unmet: ['At least 8 characters', 'At least one uppercase letter (A–Z)', 'At least one special character (...)'] }

isStrongPassword('Hunter2!')
// { valid: true, score: 5, unmet: [] }

isStrongPassword('short', { minLength: 12 })
// { valid: false, score: 0, unmet: ['At least 12 characters', ...] }

isStrongPassword('NoSpecial1', { requireSpecial: false })
// { valid: true, score: 4, unmet: [] }
```

## How it works

1. Each enabled rule is reduced to a `{ check: boolean, message: string }` entry. This keeps the scoring loop uniform regardless of which rules are active.
2. `score` is the count of passing rules — useful for driving a strength meter with `score / totalRules`.
3. `unmet` lists only the failing rule messages, ready to be rendered as a bullet list beneath a password field.

## Notes

- The `score` ceiling equals the number of active rules (up to 5 with all defaults enabled). Normalise it as `score / totalActiveRules` when you need a 0–100% meter.
- Never perform password strength checks exclusively on the client. Always enforce the same rules server-side before persisting credentials.
