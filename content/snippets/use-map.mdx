A typed hook that wraps a `Map` in React state and provides stable mutator callbacks so components that only use a single entry do not re-render on unrelated changes.

## Usage

```typescript
import { useState, useCallback } from 'react'

type MapActions<K, V> = {
  set: (key: K, value: V) => void
  delete: (key: K) => void
  clear: () => void
  reset: (entries?: Iterable<readonly [K, V]>) => void
}

function useMap<K, V>(
  initialEntries?: Iterable<readonly [K, V]>
): [Map<K, V>, MapActions<K, V>] {
  const [map, setMap] = useState<Map<K, V>>(
    () => new Map(initialEntries)
  )

  const set = useCallback((key: K, value: V) => {
    setMap(prev => new Map(prev).set(key, value))
  }, [])

  const del = useCallback((key: K) => {
    setMap(prev => {
      const next = new Map(prev)
      next.delete(key)
      return next
    })
  }, [])

  const clear = useCallback(() => {
    setMap(new Map())
  }, [])

  const reset = useCallback((entries?: Iterable<readonly [K, V]>) => {
    setMap(new Map(entries))
  }, [])

  return [map, { set, delete: del, clear, reset }]
}

export default useMap
```

## Examples

```typescript
// Track form field errors
const [errors, { set: setError, delete: clearError, clear: clearAll }] =
  useMap<string, string>()

function validate(field: string, value: string) {
  if (!value.trim()) {
    setError(field, 'This field is required')
  } else {
    clearError(field)
  }
}

// Read a single entry
const emailError = errors.get('email')
```

```typescript
// Pre-populate from an array
const [cache, { set, clear }] = useMap<number, User>(
  users.map(u => [u.id, u])
)
```

## Notes

- Every mutator creates a new `Map` instance so React detects the state change via reference equality.
- Mutator references are stable across renders (wrapped in `useCallback`) — safe to pass as props without causing child re-renders.
- `reset()` with no argument produces an empty map, equivalent to `clear()`.
