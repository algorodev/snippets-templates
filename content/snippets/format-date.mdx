Converts a `Date` object or numeric timestamp into a formatted string using familiar tokens like `YYYY-MM-DD` or `DD/MM/YYYY HH:mm`. Zero-dependency and tree-shakeable.

## Usage

```typescript
type DateInput = Date | number

/**
 * Formats a Date or timestamp using token-based patterns.
 *
 * Supported tokens:
 *   YYYY  — full year (e.g. 2024)
 *   YY    — 2-digit year (e.g. 24)
 *   MM    — zero-padded month (01–12)
 *   M     — month without padding (1–12)
 *   DD    — zero-padded day (01–31)
 *   D     — day without padding (1–31)
 *   HH    — zero-padded 24h hour (00–23)
 *   H     — hour without padding (0–23)
 *   mm    — zero-padded minutes (00–59)
 *   m     — minutes without padding (0–59)
 *   ss    — zero-padded seconds (00–59)
 *   s     — seconds without padding (0–59)
 */
function formatDate(input: DateInput, pattern: string = 'YYYY-MM-DD'): string {
  const date = typeof input === 'number' ? new Date(input) : input

  const pad = (n: number): string => String(n).padStart(2, '0')

  const year = date.getFullYear()
  const month = date.getMonth() + 1
  const day = date.getDate()
  const hours = date.getHours()
  const minutes = date.getMinutes()
  const seconds = date.getSeconds()

  const tokens: Record<string, string> = {
    YYYY: String(year),
    YY: String(year).slice(-2),
    MM: pad(month),
    M: String(month),
    DD: pad(day),
    D: String(day),
    HH: pad(hours),
    H: String(hours),
    mm: pad(minutes),
    m: String(minutes),
    ss: pad(seconds),
    s: String(seconds),
  }

  // Replace longer tokens first to avoid partial matches (e.g. MM before M)
  return pattern.replace(
    /YYYY|YY|MM|M|DD|D|HH|H|mm|m|ss|s/g,
    (token) => tokens[token]
  )
}

export default formatDate
```

## Examples

```typescript
const date = new Date('2024-03-07T09:05:42')

formatDate(date)
// => '2024-03-07'

formatDate(date, 'DD/MM/YYYY')
// => '07/03/2024'

formatDate(date, 'YYYY-MM-DD HH:mm:ss')
// => '2024-03-07 09:05:42'

formatDate(date, 'D M YYYY')
// => '7 3 2024'

// Also accepts a Unix timestamp (milliseconds)
formatDate(1709798742000, 'MM/DD/YYYY')
// => '03/07/2024'
```

## Notes

- All output is in **local time** — use `toISOString()` directly if you need UTC.
- Tokens are replaced using a single regex pass (longest first), so there are no partial-match collisions between `MM`/`M` or `DD`/`D`.
- For locale-aware formatting, timezone support, or relative time, consider `Intl.DateTimeFormat` or a library like `date-fns`.
