Wraps any sync or async function to record and report how long it took to run, without touching the function body. Pass a custom logger to route timing output wherever you need it.

## Usage

```python
import asyncio
import functools
import logging
import time
from collections.abc import Callable
from typing import Any

logger = logging.getLogger(__name__)


def timer(
    func: Callable | None = None,
    *,
    log: Callable[[str], None] | None = None,
) -> Callable:
    """Decorator that measures and logs the execution time of a function.

    Can be used with or without arguments:
        @timer
        @timer()
        @timer(log=my_logger.info)

    Args:
        func: The function being decorated (set automatically when used without parentheses).
        log:  Optional callback that receives the timing message string.
              Defaults to ``logger.debug``.
    """
    _log = log or logger.debug

    def decorator(fn: Callable) -> Callable:
        if asyncio.iscoroutinefunction(fn):
            @functools.wraps(fn)
            async def async_wrapper(*args: Any, **kwargs: Any) -> Any:
                start = time.perf_counter()
                try:
                    return await fn(*args, **kwargs)
                finally:
                    elapsed = (time.perf_counter() - start) * 1000
                    _log(f"{fn.__qualname__} completed in {elapsed:.2f} ms")
            return async_wrapper

        @functools.wraps(fn)
        def sync_wrapper(*args: Any, **kwargs: Any) -> Any:
            start = time.perf_counter()
            try:
                return fn(*args, **kwargs)
            finally:
                elapsed = (time.perf_counter() - start) * 1000
                _log(f"{fn.__qualname__} completed in {elapsed:.2f} ms")
        return sync_wrapper

    # Support bare @timer usage (no call parentheses)
    if func is not None:
        return decorator(func)
    return decorator
```

## Examples

```python
import logging
logging.basicConfig(level=logging.DEBUG)


# Bare decorator — no parentheses needed
@timer
def compute(n: int) -> int:
    total = 0
    for i in range(n):
        total += i
    return total

compute(1_000_000)
# DEBUG: compute completed in 42.17 ms


# With a custom logger callback
@timer(log=print)
def fetch_data(url: str) -> bytes:
    import urllib.request
    with urllib.request.urlopen(url) as r:
        return r.read()

fetch_data("https://example.com")
# fetch_data completed in 312.05 ms


# Async functions are handled transparently
@timer(log=print)
async def async_task(delay: float) -> str:
    await asyncio.sleep(delay)
    return "done"

asyncio.run(async_task(0.1))
# async_task completed in 100.43 ms
```

## Options

| Parameter | Type | Default | Description |
|-----------|------|---------|-------------|
| `log` | `Callable[[str], None]` | `logger.debug` | Receives the formatted timing message. Pass `print`, `logging.info`, or any callable that accepts a string. |

## Notes

- Uses `time.perf_counter()` for sub-millisecond resolution regardless of platform.
- The `finally` block ensures timing is always logged even when the decorated function raises an exception.
- `functools.wraps` preserves the original function's `__name__`, `__doc__`, and signature for introspection tools.
- The decorator supports three call styles: `@timer`, `@timer()`, and `@timer(log=...)`.
