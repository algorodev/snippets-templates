Add these functions to your `~/.bashrc` or `~/.zshrc`. They wrap `mysqldump` with sane defaults and optional S3 upload via the AWS CLI.

## Usage

```bash
# ── MySQL backup helpers ─────────────────────────────────────────────────────

# Configuration — override via environment variables
: "${MYSQL_USER:=root}"
: "${MYSQL_HOST:=127.0.0.1}"
: "${MYSQL_PORT:=3306}"
# MYSQL_PASSWORD — read from env or ~/.my.cnf; never hard-code here
: "${MYSQL_S3_BUCKET:=}"       # e.g. s3://my-bucket/mysql-backups
: "${MYSQL_BACKUP_DIR:=/tmp}"  # local staging directory

# Internal: build the auth flags for mysql/mysqldump
_mysql_auth() {
  local flags="-u$MYSQL_USER -h$MYSQL_HOST -P$MYSQL_PORT"
  [[ -n "${MYSQL_PASSWORD:-}" ]] && flags+=" -p$MYSQL_PASSWORD"
  echo "$flags"
}

# mysql-backup: dump an entire database to a timestamped .sql.gz file
# Usage: mysql-backup <dbname> [output_dir]
# Example: mysql-backup myapp
#          mysql-backup myapp /backups
mysql-backup() {
  local dbname="$1"
  local outdir="${2:-$MYSQL_BACKUP_DIR}"

  if [[ -z "$dbname" ]]; then
    echo "Usage: mysql-backup <dbname> [output_dir]"
    return 1
  fi

  mkdir -p "$outdir"
  local timestamp
  timestamp=$(date +%Y%m%d_%H%M%S)
  local outfile="$outdir/${dbname}_${timestamp}.sql.gz"

  echo "Backing up '$dbname' → $outfile ..."
  # shellcheck disable=SC2046
  mysqldump $(eval echo "$(_mysql_auth)") \
    --single-transaction \
    --routines \
    --triggers \
    --events \
    --set-gtid-purged=OFF \
    --compress \
    "$dbname" \
    | gzip -9 > "$outfile"

  local exit_code=${PIPESTATUS[0]}
  if [[ $exit_code -eq 0 ]]; then
    local size
    size=$(du -sh "$outfile" | cut -f1)
    echo "Done. File: $outfile ($size)"
  else
    echo "mysqldump failed (exit $exit_code). Removing partial file."
    rm -f "$outfile"
    return $exit_code
  fi
}

# mysql-backup-table: dump a single table to a timestamped .sql.gz file
# Usage: mysql-backup-table <dbname> <table> [output_dir]
mysql-backup-table() {
  local dbname="$1"
  local table="$2"
  local outdir="${3:-$MYSQL_BACKUP_DIR}"

  if [[ -z "$dbname" || -z "$table" ]]; then
    echo "Usage: mysql-backup-table <dbname> <table> [output_dir]"
    return 1
  fi

  mkdir -p "$outdir"
  local timestamp
  timestamp=$(date +%Y%m%d_%H%M%S)
  local outfile="$outdir/${dbname}_${table}_${timestamp}.sql.gz"

  echo "Backing up '$dbname.$table' → $outfile ..."
  # shellcheck disable=SC2046
  mysqldump $(eval echo "$(_mysql_auth)") \
    --single-transaction \
    "$dbname" "$table" \
    | gzip -9 > "$outfile"

  local exit_code=${PIPESTATUS[0]}
  if [[ $exit_code -eq 0 ]]; then
    local size
    size=$(du -sh "$outfile" | cut -f1)
    echo "Done. File: $outfile ($size)"
  else
    echo "mysqldump failed (exit $exit_code)."
    rm -f "$outfile"
    return $exit_code
  fi
}

# mysql-backup-s3: backup a database and upload to S3, then remove the local file
# Requires: AWS CLI configured with credentials that can write to MYSQL_S3_BUCKET
# Usage: mysql-backup-s3 <dbname>
mysql-backup-s3() {
  local dbname="$1"

  if [[ -z "$dbname" ]]; then
    echo "Usage: mysql-backup-s3 <dbname>"
    return 1
  fi

  if [[ -z "$MYSQL_S3_BUCKET" ]]; then
    echo "Error: MYSQL_S3_BUCKET is not set."
    return 1
  fi

  if ! command -v aws &>/dev/null; then
    echo "Error: AWS CLI not found. Install it from https://aws.amazon.com/cli/"
    return 1
  fi

  # Perform local backup first
  mysql-backup "$dbname" "$MYSQL_BACKUP_DIR" || return $?

  # Find the file we just created (most recent match)
  local outfile
  outfile=$(ls -t "$MYSQL_BACKUP_DIR/${dbname}_"*.sql.gz 2>/dev/null | head -1)
  if [[ -z "$outfile" ]]; then
    echo "Error: could not locate the backup file to upload."
    return 1
  fi

  local s3_key
  s3_key="$MYSQL_S3_BUCKET/$(basename "$outfile")"
  echo "Uploading $outfile → $s3_key ..."
  aws s3 cp "$outfile" "$s3_key" --storage-class STANDARD_IA

  local exit_code=$?
  if [[ $exit_code -eq 0 ]]; then
    echo "Upload complete. Removing local file."
    rm -f "$outfile"
  else
    echo "Upload failed (exit $exit_code). Local file kept: $outfile"
    return $exit_code
  fi
}

# mysql-restore: restore a .sql.gz dump into a database
# Usage: mysql-restore <dump_file> <dbname>
mysql-restore() {
  local dumpfile="$1"
  local dbname="$2"

  if [[ -z "$dumpfile" || -z "$dbname" ]]; then
    echo "Usage: mysql-restore <dump_file> <dbname>"
    return 1
  fi

  if [[ ! -f "$dumpfile" ]]; then
    echo "Error: file not found: $dumpfile"
    return 1
  fi

  echo "Restoring '$dumpfile' → '$dbname' ..."
  # shellcheck disable=SC2046
  gunzip -c "$dumpfile" | mysql $(eval echo "$(_mysql_auth)") "$dbname"

  local exit_code=${PIPESTATUS[1]}
  [[ $exit_code -eq 0 ]] && echo "Restore complete." || echo "mysql failed (exit $exit_code)."
  return $exit_code
}
```

## Options

| Function | Arguments | Description |
|---|---|---|
| `mysql-backup` | `<dbname> [dir]` | Full database dump, compressed to `.sql.gz` |
| `mysql-backup-table` | `<dbname> <table> [dir]` | Single-table dump, compressed to `.sql.gz` |
| `mysql-backup-s3` | `<dbname>` | Full dump then upload to `MYSQL_S3_BUCKET`, removes local file on success |
| `mysql-restore` | `<dump_file> <dbname>` | Decompress and pipe a `.sql.gz` into a database |

## Examples

```bash
# Set connection details once per session (or add to ~/.bashrc)
export MYSQL_USER=myuser MYSQL_HOST=db.example.com MYSQL_PASSWORD=secret

# Full backup to /backups
mysql-backup myapp /backups

# Back up just the users table
mysql-backup-table myapp users /backups

# Backup and ship to S3
export MYSQL_S3_BUCKET=s3://my-company-backups/mysql
mysql-backup-s3 myapp

# Restore from a dump
mysql-restore /backups/myapp_20260601_020000.sql.gz myapp_staging
```

## Notes

- `--single-transaction` makes the dump consistent for InnoDB tables without locking them for writes.
- `--routines`, `--triggers`, and `--events` ensure stored procedures and scheduled events are included in full-database backups.
- Credentials can also live in `~/.my.cnf` under `[mysqldump]` and `[client]` so `MYSQL_PASSWORD` never appears in environment or process lists.
- `STANDARD_IA` (Infrequent Access) is used for S3 uploads to reduce storage cost; change to `GLACIER` for long-term archival.
