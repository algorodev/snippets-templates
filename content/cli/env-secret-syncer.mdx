Add this function to your `~/.zshrc` or `~/.bashrc` to sync a local `.env` file to multiple cloud platforms without copy-pasting secrets. It reads key=value pairs, strips comments and blanks, and calls each platform's CLI to set the secret.

## Usage

```bash
# ── ~/.zshrc / ~/.bashrc ──────────────────────────────────────────

# Sync a .env file to one or more cloud platforms.
# Usage: envsync <platform> [--env-file <path>] [options]
#
# Platforms: vercel | fly | railway | github
# Options:
#   --env-file <path>     Path to .env file (default: .env)
#   --env <environment>   Vercel environment: production|preview|development (default: production)
#   --app <name>          Fly.io app name (default: auto-detect from fly.toml)
#   --repo <owner/repo>   GitHub repo for Actions secrets (default: git remote)
#   --dry-run             Print the keys that would be synced without making any changes

envsync() {
  local platform="${1:-}"
  local env_file=".env"
  local vercel_env="production"
  local fly_app=""
  local gh_repo=""
  local dry_run=0

  shift || true

  # Parse flags
  while [[ $# -gt 0 ]]; do
    case "$1" in
      --env-file) env_file="$2"; shift 2 ;;
      --env)      vercel_env="$2"; shift 2 ;;
      --app)      fly_app="$2"; shift 2 ;;
      --repo)     gh_repo="$2"; shift 2 ;;
      --dry-run)  dry_run=1; shift ;;
      *) echo "Unknown option: $1" >&2; return 1 ;;
    esac
  done

  # Validate platform
  if [[ -z "$platform" ]]; then
    echo "Usage: envsync <vercel|fly|railway|github> [--env-file .env] [options]" >&2
    return 1
  fi

  # Validate env file
  if [[ ! -f "$env_file" ]]; then
    echo "Error: env file not found: $env_file" >&2
    return 1
  fi

  # Read key=value pairs, skip comments and blank lines
  local -a pairs=()
  while IFS= read -r line || [[ -n "$line" ]]; do
    # Strip leading whitespace and skip comments / empty lines
    line="${line#"${line%%[! ]*}"}"
    [[ -z "$line" || "$line" == \#* ]] && continue
    [[ "$line" != *=* ]] && continue
    pairs+=("$line")
  done < "$env_file"

  if [[ ${#pairs[@]} -eq 0 ]]; then
    echo "No variables found in $env_file"
    return 0
  fi

  echo "Found ${#pairs[@]} variable(s) in $env_file"
  [[ $dry_run -eq 1 ]] && echo "[dry-run] No changes will be made."

  case "$platform" in

    vercel)
      command -v vercel &>/dev/null || { echo "Error: vercel CLI not found." >&2; return 1; }
      for pair in "${pairs[@]}"; do
        local key="${pair%%=*}"
        local val="${pair#*=}"
        # Strip surrounding quotes from the value
        val="${val%\"}" ; val="${val#\"}"
        val="${val%\'}" ; val="${val#\'}"
        if [[ $dry_run -eq 1 ]]; then
          echo "  [dry-run] vercel env add $key $vercel_env"
        else
          printf '%s' "$val" | vercel env add "$key" "$vercel_env" --yes \
            && echo "  ✓ $key → Vercel ($vercel_env)" \
            || echo "  ✗ $key failed" >&2
        fi
      done
      ;;

    fly)
      command -v flyctl &>/dev/null || { echo "Error: flyctl not found." >&2; return 1; }
      local fly_args=()
      [[ -n "$fly_app" ]] && fly_args+=("--app" "$fly_app")
      if [[ $dry_run -eq 1 ]]; then
        for pair in "${pairs[@]}"; do
          echo "  [dry-run] flyctl secrets set ${pair%%=*}=***"
        done
      else
        # flyctl accepts KEY=VALUE KEY=VALUE ... in one call
        flyctl secrets set "${pairs[@]}" "${fly_args[@]}" \
          && echo "  ✓ ${#pairs[@]} secret(s) set on Fly.io" \
          || echo "  ✗ flyctl secrets set failed" >&2
      fi
      ;;

    railway)
      command -v railway &>/dev/null || { echo "Error: railway CLI not found." >&2; return 1; }
      for pair in "${pairs[@]}"; do
        local key="${pair%%=*}"
        local val="${pair#*=}"
        val="${val%\"}" ; val="${val#\"}"
        val="${val%\'}" ; val="${val#\'}"
        if [[ $dry_run -eq 1 ]]; then
          echo "  [dry-run] railway variables set $key=***"
        else
          railway variables set "$key=$val" \
            && echo "  ✓ $key → Railway" \
            || echo "  ✗ $key failed" >&2
        fi
      done
      ;;

    github)
      command -v gh &>/dev/null || { echo "Error: gh CLI not found." >&2; return 1; }
      # Auto-detect repo from git remote if not provided
      if [[ -z "$gh_repo" ]]; then
        gh_repo=$(git remote get-url origin 2>/dev/null \
          | sed -E 's|.*github\.com[:/]||; s|\.git$||')
      fi
      if [[ -z "$gh_repo" ]]; then
        echo "Error: could not detect GitHub repo. Pass --repo owner/repo." >&2
        return 1
      fi
      echo "Target repo: $gh_repo"
      for pair in "${pairs[@]}"; do
        local key="${pair%%=*}"
        local val="${pair#*=}"
        val="${val%\"}" ; val="${val#\"}"
        val="${val%\'}" ; val="${val#\'}"
        if [[ $dry_run -eq 1 ]]; then
          echo "  [dry-run] gh secret set $key --repo $gh_repo"
        else
          printf '%s' "$val" | gh secret set "$key" --repo "$gh_repo" \
            && echo "  ✓ $key → GitHub Actions ($gh_repo)" \
            || echo "  ✗ $key failed" >&2
        fi
      done
      ;;

    *)
      echo "Error: unknown platform '$platform'. Choose: vercel | fly | railway | github" >&2
      return 1
      ;;
  esac
}
```

After adding the function, reload your shell:

```bash
source ~/.zshrc   # or source ~/.bashrc
```

## Examples

```bash
# Sync .env → Vercel production (default)
envsync vercel

# Sync .env.production → Vercel preview environment
envsync vercel --env-file .env.production --env preview

# Sync .env → Fly.io (auto-detects app from fly.toml)
envsync fly

# Sync .env → a specific Fly.io app
envsync fly --app my-app-name

# Sync .env → Railway (must be logged in and linked)
envsync railway

# Sync .env → GitHub Actions secrets (auto-detects repo from git remote)
envsync github

# Sync .env → a specific GitHub repo
envsync github --repo acme/my-repo

# Preview what would be synced without making changes
envsync vercel --dry-run
envsync github --dry-run
```

## How it works

1. The function reads the `.env` file line by line, skipping blank lines and comments (`#`).
2. It strips surrounding single or double quotes from values before sending them to platform CLIs.
3. Each platform's CLI handles the authenticated API call:
   - **Vercel** — `vercel env add KEY environment` (pipes the value via stdin to avoid shell history leaks)
   - **Fly.io** — `flyctl secrets set KEY=VALUE ...` (batched in a single call for efficiency)
   - **Railway** — `railway variables set KEY=VALUE` (one call per variable)
   - **GitHub** — `gh secret set KEY` (pipes the value via stdin)

## Notes

- Secrets are piped via stdin or passed as arguments to avoid leaking values into shell history or process listings.
- Variables already set on the platform will be overwritten. Use `--dry-run` to inspect what will change before committing.
- The function does not delete variables removed from your `.env`. Clean up stale secrets manually through each platform's dashboard or CLI.
- For GitHub, only **Actions secrets** are set. Use `gh secret set --env <environment>` manually for environment-scoped secrets.
- Fly.io's `flyctl secrets set` triggers a rolling machine restart by default. Pass `--stage` to stage secrets without restarting.
