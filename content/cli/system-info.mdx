A single shell function that assembles the most useful system vitals — OS, hardware, uptime, network, and top processes — into a readable summary. Works on both macOS and Linux without any third-party dependencies.

## Usage

```bash
# Add to ~/.zshrc or ~/.bashrc, then reload
source ~/.zshrc

# Print the system summary
sysinfo

# Alias for quick access
alias si='sysinfo'
```

## Shell function

```bash
sysinfo() {
  local os_name os_version cpu_model cpu_cores ram_total ram_used
  local uptime_str hostname_str kernel_str

  # ── Detect platform ────────────────────────────────────────────────────────
  local platform
  platform=$(uname -s)

  # ── Header ─────────────────────────────────────────────────────────────────
  hostname_str=$(hostname -f 2>/dev/null || hostname)
  kernel_str=$(uname -r)

  printf '\n\033[1;34m══ System Info: %s ══\033[0m\n\n' "$hostname_str"

  # ── OS ─────────────────────────────────────────────────────────────────────
  if [[ "$platform" == "Darwin" ]]; then
    os_name="macOS"
    os_version=$(sw_vers -productVersion)
    local build
    build=$(sw_vers -buildVersion)
    printf '  \033[1mOS         :\033[0m %s %s (build %s)\n' "$os_name" "$os_version" "$build"
  else
    if [[ -f /etc/os-release ]]; then
      # shellcheck source=/dev/null
      source /etc/os-release
      os_name="${NAME:-Linux}"
      os_version="${VERSION_ID:-unknown}"
    else
      os_name="Linux"
      os_version="unknown"
    fi
    printf '  \033[1mOS         :\033[0m %s %s\n' "$os_name" "$os_version"
  fi
  printf '  \033[1mKernel     :\033[0m %s\n' "$kernel_str"

  # ── CPU ────────────────────────────────────────────────────────────────────
  if [[ "$platform" == "Darwin" ]]; then
    cpu_model=$(sysctl -n machdep.cpu.brand_string 2>/dev/null || echo "unknown")
    cpu_cores=$(sysctl -n hw.logicalcpu 2>/dev/null || echo "?")
  else
    cpu_model=$(grep -m1 'model name' /proc/cpuinfo 2>/dev/null \
      | cut -d: -f2 | xargs || echo "unknown")
    cpu_cores=$(nproc 2>/dev/null || grep -c '^processor' /proc/cpuinfo 2>/dev/null || echo "?")
  fi
  printf '  \033[1mCPU        :\033[0m %s (%s logical cores)\n' "$cpu_model" "$cpu_cores"

  # ── RAM ────────────────────────────────────────────────────────────────────
  if [[ "$platform" == "Darwin" ]]; then
    local ram_bytes
    ram_bytes=$(sysctl -n hw.memsize 2>/dev/null || echo 0)
    ram_total=$(( ram_bytes / 1024 / 1024 / 1024 ))
    # macOS memory pressure via vm_stat
    local page_size pages_free pages_spec pages_inactive pages_active pages_wired
    page_size=$(vm_stat | awk '/page size/ {print $8}')
    pages_free=$(vm_stat | awk '/Pages free/ {gsub(/\./,"",$3); print $3}')
    pages_spec=$(vm_stat | awk '/Pages speculative/ {gsub(/\./,"",$3); print $3}')
    pages_wired=$(vm_stat | awk '/Pages wired down/ {gsub(/\./,"",$4); print $4}')
    pages_active=$(vm_stat | awk '/Pages active/ {gsub(/\./,"",$3); print $3}')
    pages_free=$(( (pages_free + pages_spec) * page_size / 1024 / 1024 / 1024 ))
    ram_used=$(( ram_total - pages_free ))
    printf '  \033[1mRAM        :\033[0m %dGB used / %dGB total\n' "$ram_used" "$ram_total"
  else
    local mem_total mem_available
    mem_total=$(awk '/MemTotal/ {print $2}' /proc/meminfo)
    mem_available=$(awk '/MemAvailable/ {print $2}' /proc/meminfo)
    ram_total=$(( mem_total / 1024 ))
    ram_used=$(( (mem_total - mem_available) / 1024 ))
    printf '  \033[1mRAM        :\033[0m %dMB used / %dMB total\n' "$ram_used" "$ram_total"
  fi

  # ── Disk ───────────────────────────────────────────────────────────────────
  printf '  \033[1mDisk (/)   :\033[0m '
  df -h / 2>/dev/null | awk 'NR==2 {printf "%s used / %s total (%s)\n", $3, $2, $5}'

  # ── Uptime ─────────────────────────────────────────────────────────────────
  if command -v uptime &>/dev/null; then
    uptime_str=$(uptime | sed 's/.*up /up /' | sed 's/,.*//')
    printf '  \033[1mUptime     :\033[0m %s\n' "$uptime_str"
  fi

  # ── Load average ──────────────────────────────────────────────────────────
  local load1 load5 load15
  if [[ "$platform" == "Darwin" ]]; then
    read -r load1 load5 load15 < <(sysctl -n vm.loadavg | tr -d '{}' | awk '{print $1, $2, $3}')
  else
    read -r load1 load5 load15 < <(cut -d' ' -f1-3 /proc/loadavg)
  fi
  printf '  \033[1mLoad avg   :\033[0m %s %s %s (1/5/15 min)\n' "$load1" "$load5" "$load15"

  # ── IP addresses ──────────────────────────────────────────────────────────
  printf '\n  \033[1mNetwork:\033[0m\n'
  if [[ "$platform" == "Darwin" ]]; then
    networksetup -listallhardwareports 2>/dev/null \
      | awk '/Hardware Port/{port=$NF} /Device:/{dev=$2} /Ethernet Address:/{mac=$NF; print port, dev, mac}' \
      | while read -r port dev mac; do
          local ip
          ip=$(ipconfig getifaddr "$dev" 2>/dev/null || echo "—")
          printf '    %-12s %-10s ip=%-18s mac=%s\n' "$port" "$dev" "$ip" "$mac"
        done
  else
    ip -o -4 addr show 2>/dev/null \
      | awk '{printf "    %-12s %s\n", $2, $4}' \
      | grep -v '127.0.0.1'
  fi

  # ── Top CPU consumers ─────────────────────────────────────────────────────
  printf '\n  \033[1mTop 5 by CPU:\033[0m\n'
  if [[ "$platform" == "Darwin" ]]; then
    ps -Acro %cpu,pid,comm 2>/dev/null | head -6 | tail -5 \
      | awk '{printf "    %6s%%  pid=%-8s %s\n", $1, $2, $3}'
  else
    ps -eo %cpu,pid,comm --sort=-%cpu 2>/dev/null | head -6 | tail -5 \
      | awk '{printf "    %6s%%  pid=%-8s %s\n", $1, $2, $3}'
  fi

  # ── Top memory consumers ──────────────────────────────────────────────────
  printf '\n  \033[1mTop 5 by RAM:\033[0m\n'
  if [[ "$platform" == "Darwin" ]]; then
    ps -Acro %mem,pid,comm 2>/dev/null | head -6 | tail -5 \
      | awk '{printf "    %6s%%  pid=%-8s %s\n", $1, $2, $3}'
  else
    ps -eo %mem,pid,comm --sort=-%mem 2>/dev/null | head -6 | tail -5 \
      | awk '{printf "    %6s%%  pid=%-8s %s\n", $1, $2, $3}'
  fi

  printf '\n'
}
```

## Examples

```bash
# Quick alias
alias si='sysinfo'

# Pipe to less for easy scrolling on busy servers
sysinfo | less -R

# Log to a file for later comparison
sysinfo >> ~/sysinfo-$(date +%F).log

# Run periodically via watch (GNU/Linux only)
watch -n 5 bash -c 'source ~/.bashrc && sysinfo'
```

## Notes

- All output uses ANSI colour codes. If your terminal does not support colour, strip them with `sysinfo | sed 's/\x1b\[[0-9;]*m//g'`.
- On macOS, network information relies on `networksetup`, which ships with the OS. No Homebrew packages are required.
- On Linux, `/proc/cpuinfo`, `/proc/meminfo`, and the `ip` command (from `iproute2`) are used. These are standard on all major distributions.
- RAM reporting on macOS uses `vm_stat` page counts rather than `top` or Activity Monitor to stay scriptable and fast.
