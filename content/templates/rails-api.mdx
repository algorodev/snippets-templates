A Rails 7 API-only app that wires Devise + JWT for token auth, Active Model Serializers for consistent JSON, RSpec with FactoryBot for tests, and a Docker Compose stack so the whole team shares the same PostgreSQL version.

## Usage

```ruby
# config/routes.rb
Rails.application.routes.draw do
  namespace :api do
    namespace :v1 do
      devise_for :users,
                 path: '',
                 path_names: { sign_in: 'login', sign_out: 'logout', registration: 'signup' },
                 controllers: {
                   sessions:      'api/v1/sessions',
                   registrations: 'api/v1/registrations'
                 }

      resources :articles, only: %i[index show create update destroy]
    end
  end
end
```

```ruby
# app/models/user.rb
class User < ApplicationRecord
  devise :database_authenticatable, :registerable,
         :recoverable, :rememberable, :validatable,
         :jwt_authenticatable, jwt_revocation_strategy: JwtDenylist

  has_many :articles, dependent: :destroy
end
```

```ruby
# app/models/jwt_denylist.rb
class JwtDenylist < ApplicationRecord
  include Devise::JWT::RevocationStrategies::Denylist

  self.table_name = 'jwt_denylist'
end
```

```ruby
# app/models/article.rb
class Article < ApplicationRecord
  belongs_to :user

  validates :title, presence: true, length: { maximum: 255 }
  validates :body,  presence: true

  scope :published, -> { where.not(published_at: nil) }
  scope :recent,    -> { order(created_at: :desc) }
end
```

```ruby
# app/controllers/api/v1/registrations_controller.rb
module Api
  module V1
    class RegistrationsController < Devise::RegistrationsController
      respond_to :json

      private

      def respond_with(resource, _opts = {})
        if resource.persisted?
          render json: { message: 'Signed up successfully.', user: UserSerializer.new(resource) }, status: :created
        else
          render json: { errors: resource.errors.full_messages }, status: :unprocessable_entity
        end
      end
    end
  end
end
```

```ruby
# app/controllers/api/v1/sessions_controller.rb
module Api
  module V1
    class SessionsController < Devise::SessionsController
      respond_to :json

      private

      def respond_with(resource, _opts = {})
        render json: { message: 'Logged in.', user: UserSerializer.new(resource) }
      end

      def respond_to_on_destroy
        render json: { message: 'Logged out.' }
      end
    end
  end
end
```

```ruby
# app/controllers/api/v1/articles_controller.rb
module Api
  module V1
    class ArticlesController < ApplicationController
      before_action :authenticate_user!
      before_action :set_article, only: %i[show update destroy]

      def index
        articles = current_user.articles.recent
        render json: articles, each_serializer: ArticleSerializer
      end

      def show
        render json: @article, serializer: ArticleSerializer
      end

      def create
        article = current_user.articles.build(article_params)
        if article.save
          render json: article, serializer: ArticleSerializer, status: :created
        else
          render json: { errors: article.errors.full_messages }, status: :unprocessable_entity
        end
      end

      def update
        if @article.update(article_params)
          render json: @article, serializer: ArticleSerializer
        else
          render json: { errors: @article.errors.full_messages }, status: :unprocessable_entity
        end
      end

      def destroy
        @article.destroy
        head :no_content
      end

      private

      def set_article
        @article = current_user.articles.find(params[:id])
      rescue ActiveRecord::RecordNotFound
        render json: { error: 'Article not found' }, status: :not_found
      end

      def article_params
        params.require(:article).permit(:title, :body, :published_at)
      end
    end
  end
end
```

```ruby
  belongs_to :user, serializer: UserSerializer
end
```

```ruby
# spec/requests/api/v1/articles_spec.rb
require 'rails_helper'

RSpec.describe 'Api::V1::Articles', type: :request do
  let(:user)    { create(:user) }
  let(:headers) { user.create_new_auth_token }

  describe 'GET /api/v1/articles' do
    it 'returns 401 without auth' do
      get '/api/v1/articles'
      expect(response).to have_http_status(:unauthorized)
    end

    it 'returns the user articles' do
      create_list(:article, 3, user: user)
      get '/api/v1/articles', headers: auth_headers(user)
      expect(response).to have_http_status(:ok)
      expect(json['data'].size).to eq(3)
    end
  end

  describe 'POST /api/v1/articles' do
    it 'creates an article' do
      post '/api/v1/articles',
           params: { article: { title: 'Test', body: 'Body text' } },
           headers: auth_headers(user)
      expect(response).to have_http_status(:created)
      expect(json['title']).to eq('Test')
    end

    it 'returns errors for invalid params' do
      post '/api/v1/articles',
           params: { article: { title: '' } },
           headers: auth_headers(user)
      expect(response).to have_http_status(:unprocessable_entity)
    end
  end

  describe 'DELETE /api/v1/articles/:id' do
    it 'cannot delete another user article' do
      other_article = create(:article)
      delete "/api/v1/articles/#{other_article.id}", headers: auth_headers(user)
      expect(response).to have_http_status(:not_found)
    end
  end

  private

  def auth_headers(user)
    post '/api/v1/users/sign_in',
         params: { user: { email: user.email, password: user.password } }
    { 'Authorization' => "Bearer #{json['token']}" }
  end

  def json
    JSON.parse(response.body)
  end
end
```

## How it works

`devise-jwt` hooks into Devise's Warden strategy to sign a JWT on login and place it in the `Authorization` response header. Subsequent requests must include `Authorization: Bearer <token>`. On logout the token is written to the `jwt_denylist` table, immediately invalidating it even before expiry. Active Model Serializers intercept `render json:` calls and transform the object using the matching `*Serializer` class, keeping controller code clean.

## Notes

- Set `config.jwt_secret` in `config/initializers/devise.rb` to `Rails.application.credentials.jwt_secret_key`.
- Replace the denylist strategy with `JTIMatcher` for better performance at scale (trades revocation granularity for a DB query per request).
- Add `rack-cors` to `Gemfile` and configure `config/initializers/cors.rb` before deploying behind a browser client.

## README

### What this includes

- Rails 7 API-only application (`--api` mode)
- Devise with `devise-jwt` for stateless Bearer token authentication
- `JwtDenylist` model for immediate token revocation on logout
- `Article` model with `belongs_to :user` and scopes
- Active Model Serializers for JSON response shaping
- Versioned routes under `/api/v1/`
- RSpec with FactoryBot and `rails-controller-testing`
- Docker Compose with PostgreSQL 16

### Prerequisites

- Ruby 3.3+
- Bundler: `gem install bundler`
- Docker Desktop (for the Compose stack)

### Create this template

```bash
# 1. Create a new API-only Rails app
rails new my-rails-api --api --database=postgresql
cd my-rails-api

# 2. Add gems to Gemfile
# gem 'devise'
# gem 'devise-jwt'
# gem 'active_model_serializers'
# group :development, :test do
#   gem 'rspec-rails'
#   gem 'factory_bot_rails'
#   gem 'faker'
# end

bundle install

# 3. Install RSpec
rails generate rspec:install

# 4. Install Devise and generate User
rails generate devise:install
rails generate devise User
rails generate model JwtDenylist jti:string:index exp:datetime
rails db:migrate

# 5. Generate Article resource
rails generate model Article user:references title:string body:text published_at:datetime
rails generate controller Api::V1::Articles --no-helper --no-assets --skip-routes
rails db:migrate

# 6. Generate serializers
rails generate serializer Article title body published_at created_at updated_at
rails generate serializer User email created_at

# 7. Add credentials for JWT secret
rails credentials:edit
# Add: jwt_secret_key: <run `rails secret` to generate>

# 8. Copy routes, models, controllers, serializers as shown above

# 9. Start with Docker Compose
docker compose up
# or without Docker:
rails db:prepare && rails server
# API base: http://localhost:3000/api/v1
```

### First things to customize

- `config/credentials.yml.enc` — add `jwt_secret_key` with a long random string
- `app/models/article.rb` — swap `Article` for your own domain model
- `config/routes.rb` — add or rename `resources` blocks under `api/v1`
- `app/serializers/` — add/remove attributes to match your API contract
- `docker-compose.yml` `POSTGRES_*` env vars — change credentials before deploying
- JWT expiration: set `config.jwt.expiration_time` in the Devise initializer
