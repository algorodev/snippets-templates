A full-stack storefront built on Next.js App Router that handles product browsing, a persistent cart, and Stripe Checkout — all wired together with Server Actions and a headless CMS for product data.

## Usage

```typescript
// lib/stripe.ts
import Stripe from 'stripe'

export const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, {
  apiVersion: '2025-04-30.basil',
})
```

```typescript
// lib/cart.ts  (server-side cart stored in a signed cookie)
import { cookies } from 'next/headers'
import { cache } from 'react'

export interface CartItem {
  productId: string
  variantId: string
  quantity: number
  price: number
  title: string
  image: string
}

export interface Cart {
  items: CartItem[]
  total: number
}

export const getCart = cache(async (): Promise<Cart> => {
  const cookieStore = await cookies()
  const raw = cookieStore.get('cart')?.value
  if (!raw) return { items: [], total: 0 }
  try {
    const items: CartItem[] = JSON.parse(Buffer.from(raw, 'base64').toString())
    const total = items.reduce((sum, i) => sum + i.price * i.quantity, 0)
    return { items, total }
  } catch {
    return { items: [], total: 0 }
  }
})
```

```typescript
// app/actions/cart.ts
'use server'

import { cookies } from 'next/headers'
import { revalidatePath } from 'next/cache'
import { type CartItem } from '@/lib/cart'

export async function addToCart(item: CartItem) {
  const cookieStore = await cookies()
  const raw = cookieStore.get('cart')?.value
  const items: CartItem[] = raw
    ? JSON.parse(Buffer.from(raw, 'base64').toString())
    : []

  const existing = items.find(
    (i) => i.productId === item.productId && i.variantId === item.variantId,
  )
  if (existing) {
    existing.quantity += item.quantity
  } else {
    items.push(item)
  }

  cookieStore.set('cart', Buffer.from(JSON.stringify(items)).toString('base64'), {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: 60 * 60 * 24 * 30, // 30 days
  })

  revalidatePath('/cart')
}

export async function removeFromCart(productId: string, variantId: string) {
  const cookieStore = await cookies()
  const raw = cookieStore.get('cart')?.value
  if (!raw) return

  const items: CartItem[] = JSON.parse(Buffer.from(raw, 'base64').toString())
  const updated = items.filter(
    (i) => !(i.productId === productId && i.variantId === variantId),
  )

  cookieStore.set('cart', Buffer.from(JSON.stringify(updated)).toString('base64'), {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: 60 * 60 * 24 * 30,
  })

  revalidatePath('/cart')
}
```

```typescript
// app/actions/checkout.ts
'use server'

import { redirect } from 'next/navigation'
import { stripe } from '@/lib/stripe'
import { getCart } from '@/lib/cart'

export async function createCheckoutSession() {
  const cart = await getCart()
  if (cart.items.length === 0) return

  const session = await stripe.checkout.sessions.create({
    mode: 'payment',
    line_items: cart.items.map((item) => ({
      price_data: {
        currency: 'usd',
        product_data: {
          name: item.title,
          images: [item.image],
        },
        unit_amount: Math.round(item.price * 100), // cents
      },
      quantity: item.quantity,
    })),
    success_url: `${process.env.NEXT_PUBLIC_BASE_URL}/order/success?session_id={CHECKOUT_SESSION_ID}`,
    cancel_url: `${process.env.NEXT_PUBLIC_BASE_URL}/cart`,
  })

  redirect(session.url!)
}
```

```typescript
// app/api/webhooks/stripe/route.ts
import { NextRequest, NextResponse } from 'next/server'
import Stripe from 'stripe'
import { stripe } from '@/lib/stripe'

export async function POST(req: NextRequest) {
  const body = await req.text()
  const sig = req.headers.get('stripe-signature')!

  let event: Stripe.Event
  try {
    event = stripe.webhooks.constructEvent(
      body,
      sig,
      process.env.STRIPE_WEBHOOK_SECRET!,
    )
  } catch {
    return NextResponse.json({ error: 'Invalid signature' }, { status: 400 })
  }

  if (event.type === 'checkout.session.completed') {
    const session = event.data.object as Stripe.Checkout.Session
    // Fulfill order: update DB, send confirmation email, etc.
    console.log('Order fulfilled for session:', session.id)
  }

  return NextResponse.json({ received: true })
}
```

```typescript
// app/products/[slug]/page.tsx
import { notFound } from 'next/navigation'
import { getProduct, getProducts } from '@/lib/products'
import { AddToCartButton } from '@/components/add-to-cart-button'

interface PageProps {
  params: Promise<{ slug: string }>
}

export async function generateStaticParams() {
  const products = await getProducts()
  return products.map((p) => ({ slug: p.slug }))
}

export default async function ProductPage({ params }: PageProps) {
  const { slug } = await params
  const product = await getProduct(slug)
  if (!product) notFound()

  return (
    <div className="mx-auto max-w-5xl px-4 py-12">
      <div className="grid gap-12 md:grid-cols-2">
        <img
          src={product.images[0]}
          alt={product.title}
          className="aspect-square w-full rounded-xl object-cover"
        />
        <div className="flex flex-col gap-4">
          <h1 className="text-3xl font-bold">{product.title}</h1>
          <p className="text-2xl font-semibold">${product.price.toFixed(2)}</p>
          <p className="text-slate-600">{product.description}</p>
          <AddToCartButton product={product} />
        </div>
      </div>
    </div>
  )
}
```

## Options

| Variable | Description |
|----------|-------------|
| `STRIPE_SECRET_KEY` | Stripe secret key (server-only) |
| `STRIPE_WEBHOOK_SECRET` | Webhook signing secret from Stripe CLI or dashboard |
| `NEXT_PUBLIC_BASE_URL` | Full origin URL for redirect URLs (e.g. `https://mystore.com`) |
| `NEXT_PUBLIC_STRIPE_PUBLISHABLE_KEY` | Publishable key for Stripe.js (client-side) |

## Notes

The cart is stored in a signed, `httpOnly` cookie so no database is required for the cart state. Swap `lib/products.ts` to point at your headless CMS (Shopify Storefront API, Sanity, Contentful, etc.) — the Server Actions and checkout flow stay unchanged. Listen to Stripe webhooks locally with `stripe listen --forward-to localhost:3000/api/webhooks/stripe`.

## README

### What this includes
- Next.js App Router with React Server Components for product pages
- Server Actions for add-to-cart and remove-from-cart (no API routes needed)
- Cookie-based cart (httpOnly, 30-day expiry)
- Stripe Checkout with line items built from the cart
- Stripe webhook handler to fulfill orders after payment
- `generateStaticParams` for all product pages (ISR-compatible)
- Tailwind CSS for styling

### Prerequisites
- Node.js 20.9+
- A [Stripe account](https://stripe.com) (free)
- A headless CMS or static product data source

### Create this template

```bash
npx create-next-app@latest my-store --typescript --tailwind --eslint --app --yes
cd my-store
npm install stripe
npm install -D @types/node
```

Set up environment variables:

```bash
cp .env.example .env.local
# Fill in STRIPE_SECRET_KEY, STRIPE_WEBHOOK_SECRET, NEXT_PUBLIC_BASE_URL
```

```bash
# Install Stripe CLI for local webhook forwarding (macOS)
brew install stripe/stripe-cli/stripe
stripe login
stripe listen --forward-to localhost:3000/api/webhooks/stripe
```

```bash
npm run dev
```

### First things to customize
- `lib/products.ts` — connect to your real product catalog (Shopify, Sanity, Contentful, or a local JSON file)
- `.env.local` — add your Stripe keys and `NEXT_PUBLIC_BASE_URL`
- `app/api/webhooks/stripe/route.ts` — add order fulfillment logic (write to DB, send email)
- `app/layout.tsx` — update store name, logo, and color scheme
- `components/add-to-cart-button.tsx` — add variant selectors (size, color) to match your product model
