A Qwik City application that ships near-zero JavaScript on page load through resumability — the server serializes component state into HTML and the browser resumes execution exactly where the server left off, without replaying event handlers.

## Usage

```typescript
// src/routes/layout.tsx
import { component$, Slot } from '@builder.io/qwik'
import { routeLoader$ } from '@builder.io/qwik-city'
import type { RequestHandler } from '@builder.io/qwik-city'
import Header from '~/components/header/header'
import Footer from '~/components/footer/footer'

export const onGet: RequestHandler = async ({ cacheControl }) => {
  cacheControl({ staleWhileRevalidate: 60 * 60 * 24 * 7, maxAge: 5 })
}

export const useServerTimeLoader = routeLoader$(() => {
  return { date: new Date().toISOString() }
})

export default component$(() => {
  return (
    <>
      <Header />
      <main>
        <Slot />
      </main>
      <Footer />
    </>
  )
})
```

```typescript
// src/routes/index.tsx
import { component$ } from '@builder.io/qwik'
import { routeLoader$, Form, routeAction$, zod$, z } from '@builder.io/qwik-city'
import type { DocumentHead } from '@builder.io/qwik-city'

// routeLoader$ runs on the server; result is serialized into HTML
export const usePostsLoader = routeLoader$(async () => {
  const res = await fetch('https://jsonplaceholder.typicode.com/posts?_limit=10')
  const posts = (await res.json()) as Array<{ id: number; title: string; body: string }>
  return posts
})

// routeAction$ handles form submissions with full type safety
export const useAddComment = routeAction$(
  async (data, { fail }) => {
    // Replace with a real DB call
    if (data.comment.length < 5) {
      return fail(400, { message: 'Comment must be at least 5 characters.' })
    }
    return { success: true }
  },
  zod$({ comment: z.string().min(1) })
)

export default component$(() => {
  const posts = usePostsLoader()
  const addComment = useAddComment()

  return (
    <div class="mx-auto max-w-2xl px-4 py-12">
      <h1 class="mb-8 text-3xl font-bold">Latest Posts</h1>

      <ul class="mb-12 space-y-6">
        {posts.value.map((post) => (
          <li key={post.id} class="rounded-lg border p-6">
            <h2 class="mb-2 text-lg font-semibold capitalize">{post.title}</h2>
            <p class="text-sm text-neutral-600">{post.body}</p>
          </li>
        ))}
      </ul>

      <section>
        <h2 class="mb-4 text-xl font-semibold">Leave a comment</h2>
        <Form action={addComment} class="space-y-4">
          <textarea
            name="comment"
            rows={4}
            placeholder="Write your comment…"
            class="block w-full rounded-md border px-3 py-2 text-sm"
          />
          {addComment.value?.failed && (
            <p class="text-sm text-red-600">{addComment.value.message}</p>
          )}
          {addComment.value?.success && (
            <p class="text-sm text-green-600">Comment submitted!</p>
          )}
          <button
            type="submit"
            class="rounded-md bg-blue-600 px-4 py-2 text-sm font-semibold text-white hover:bg-blue-700"
          >
            Submit
          </button>
        </Form>
      </section>
    </div>
  )
})

export const head: DocumentHead = {
  title: 'Qwik App',
  meta: [{ name: 'description', content: 'A resumable Qwik application.' }],
}
```

```typescript
// src/components/counter/counter.tsx
// $ suffix marks lazy boundaries — the onClick$ handler is NOT included
// in the initial HTML payload. It is downloaded only when the user clicks.
import { component$, useSignal } from '@builder.io/qwik'

export default component$(() => {
  const count = useSignal(0)

  return (
    <div class="flex items-center gap-4">
      <button
        class="rounded-md border px-3 py-1 text-sm hover:bg-neutral-100"
        onClick$={() => count.value--}
      >
        −
      </button>
      <span class="tabular-nums">{count.value}</span>
      <button
        class="rounded-md border px-3 py-1 text-sm hover:bg-neutral-100"
        onClick$={() => count.value++}
      >
        +
      </button>
    </div>
  )
})
```

```typescript
// src/routes/api/items/index.ts
// Qwik City endpoint — runs only on the server
import type { RequestHandler } from '@builder.io/qwik-city'

const items = [
  { id: 1, name: 'Apple' },
  { id: 2, name: 'Banana' },
]

export const onGet: RequestHandler = ({ json }) => {
  json(200, items)
}

export const onPost: RequestHandler = async ({ json, request }) => {
  const body = (await request.json()) as { name: string }
  const item = { id: items.length + 1, name: body.name }
  items.push(item)
  json(201, item)
}
```

## How it works

Qwik serializes the entire application state — component props, signals, and event handler references — as compressed JSON inside `<script type="qwik/json">` tags in the initial HTML. When the user interacts, the browser deserializes only the tiny slice of state needed for that specific handler, downloads the handler's chunk on demand, and continues execution. No hydration waterfall, no full component tree re-evaluation.

## README

### What this includes

- **Resumability** — server state serialized into HTML; browser resumes without replaying
- **Qwik City** — file-based routing, layouts, `routeLoader$`, `routeAction$`, and API endpoints
- **Fine-grained lazy loading** — every `$`-suffixed closure becomes an independently loadable chunk
- **`routeLoader$`** — type-safe server data fetching that serializes results into HTML
- **`routeAction$` + Zod** — form action handler with schema validation via `zod$`
- **Signals** — reactive primitives (`useSignal`, `useStore`) for local UI state
- **Server endpoints** — `onGet`/`onPost` request handlers for JSON APIs
- **Tailwind CSS** — utility-first styling via `@qwikdev/astro` or Vite plugin

### Prerequisites

- Node.js 20.9 or later
- npm, pnpm, yarn, or bun

### Create this template

```bash
# 1. Scaffold a new Qwik City app
npm create qwik@latest my-app
# Choose: Empty App, TypeScript, install dependencies

cd my-app

# 2. Add Tailwind CSS
npm run qwik add tailwind

# 3. (Optional) Add a Qwik City adapter for your deploy target
#    Cloudflare Pages:
npm run qwik add cloudflare-pages
#    Node.js / Express:
npm run qwik add express
#    Vercel Edge:
npm run qwik add vercel-edge

# 4. Start the dev server
npm run dev
```

### First things to customize

- `src/root.tsx` — update the site name and global `<head>` tags
- `src/routes/layout.tsx` — change the nav links and page shell
- `src/global.css` — add Tailwind base styles and custom CSS variables
- `src/routes/index.tsx` — replace the placeholder `routeLoader$` with a real data source
- `vite.config.ts` — configure path aliases, environment variables, and plugins
- `adapters/` — pick and configure the right adapter for your hosting platform (Vercel, Cloudflare, Node)
