The go-to scaffold for production Go CLIs — Cobra handles subcommands and flag parsing, Viper manages config files and environment variables, and zerolog gives you fast, structured logging with zero overhead when disabled.

## Usage

```bash
# Build and run
go build -o mycli .
./mycli --help
./mycli greet --name Alice
./mycli version
```

## Main entry point

```go
// main.go
package main

import "github.com/your-org/mycli/cmd"

func main() {
	cmd.Execute()
}
```

## Root command

```go
// cmd/root.go
package cmd

import (
	"os"

	"github.com/rs/zerolog"
	"github.com/rs/zerolog/log"
	"github.com/spf13/cobra"
	"github.com/spf13/viper"
)

var (
	cfgFile string
	verbose bool
)

var rootCmd = &cobra.Command{
	Use:   "mycli",
	Short: "A brief description of your CLI",
	Long:  `A longer description of your CLI application.`,
	PersistentPreRun: func(cmd *cobra.Command, args []string) {
		if verbose {
			zerolog.SetGlobalLevel(zerolog.DebugLevel)
		} else {
			zerolog.SetGlobalLevel(zerolog.InfoLevel)
		}
	},
}

func Execute() {
	if err := rootCmd.Execute(); err != nil {
		os.Exit(1)
	}
}

func init() {
	cobra.OnInitialize(initConfig)

	rootCmd.PersistentFlags().StringVar(&cfgFile, "config", "", "config file (default: $HOME/.mycli.yaml)")
	rootCmd.PersistentFlags().BoolVarP(&verbose, "verbose", "v", false, "enable debug logging")

	log.Logger = log.Output(zerolog.ConsoleWriter{Out: os.Stderr})
}

func initConfig() {
	if cfgFile != "" {
		viper.SetConfigFile(cfgFile)
	} else {
		home, err := os.UserHomeDir()
		cobra.CheckErr(err)
		viper.AddConfigPath(home)
		viper.SetConfigType("yaml")
		viper.SetConfigName(".mycli")
	}

	viper.AutomaticEnv()
	viper.SetEnvPrefix("MYCLI")

	if err := viper.ReadInConfig(); err == nil {
		log.Debug().Str("config", viper.ConfigFileUsed()).Msg("config loaded")
	}
}
```

## Subcommand example

```go
// cmd/greet.go
package cmd

import (
	"github.com/rs/zerolog/log"
	"github.com/spf13/cobra"
	"github.com/spf13/viper"
)

var greetCmd = &cobra.Command{
	Use:   "greet",
	Short: "Greet a user by name",
	Long:  `Prints a greeting message. The name can be set via flag, config file, or MYCLI_NAME env var.`,
	RunE: func(cmd *cobra.Command, args []string) error {
		name := viper.GetString("name")
		log.Info().Str("name", name).Msg("greeting user")
		cmd.Printf("Hello, %s!\n", name)
		return nil
	},
}

func init() {
	rootCmd.AddCommand(greetCmd)

	greetCmd.Flags().String("name", "World", "name to greet")
	viper.BindPFlag("name", greetCmd.Flags().Lookup("name"))
}
```

## Version command

```go
// cmd/version.go
package cmd

import (
	"fmt"

	"github.com/spf13/cobra"
)

var (
	Version   = "dev"
	Commit    = "none"
	BuildDate = "unknown"
)

var versionCmd = &cobra.Command{
	Use:   "version",
	Short: "Print version information",
	Run: func(cmd *cobra.Command, args []string) {
		fmt.Printf("mycli %s (commit: %s, built: %s)\n", Version, Commit, BuildDate)
	},
}

func init() {
	rootCmd.AddCommand(versionCmd)
}
```

## Build with version injection

```bash
go build \
  -ldflags "-X 'github.com/your-org/mycli/cmd.Version=1.0.0' \
            -X 'github.com/your-org/mycli/cmd.Commit=$(git rev-parse --short HEAD)' \
            -X 'github.com/your-org/mycli/cmd.BuildDate=$(date -u +%Y-%m-%dT%H:%M:%SZ)'" \
  -o mycli .
```

## README

### What this includes

- **Cobra** — Subcommand routing, persistent/local flags, help generation, and shell completions
- **Viper** — Layered config: flags > env vars > config file > defaults, all resolved automatically
- **zerolog** — Structured JSON logging with a human-friendly console writer for development
- **Version injection** — Build-time version, commit, and date via `-ldflags`
- **Config file support** — `~/.mycli.yaml` loaded automatically; overridable via `--config`
- **`PersistentPreRun`** — Global setup (log level) that applies to every subcommand

### Prerequisites

- Go 1.22+
- `git` (for commit hash in version builds)

### Create this template

```bash
# 1. Create the module
mkdir mycli && cd mycli
go mod init github.com/your-org/mycli

# 2. Install dependencies
go get github.com/spf13/cobra@latest
go get github.com/spf13/viper@latest
go get github.com/rs/zerolog@latest

# 3. Scaffold the cmd package
mkdir cmd

# 4. Create the files shown above:
#    main.go, cmd/root.go, cmd/greet.go, cmd/version.go

# 5. Tidy and build
go mod tidy
go build -o mycli .

# 6. Verify
./mycli --help
./mycli greet --name Alice
./mycli version
```

### Project structure

```
mycli/
├── cmd/
│   ├── root.go        # Root command, Viper init, persistent flags
│   ├── greet.go       # Example subcommand
│   └── version.go     # Version subcommand with build-time injection
├── internal/          # Private packages (add as needed)
├── main.go            # Entry point — calls cmd.Execute()
├── go.mod
└── go.sum
```

### First things to customize

- `cmd/root.go` — Replace `mycli` with your binary name in `Use`, `Short`, and `Long`
- `cmd/root.go` — Change the `viper.SetEnvPrefix("MYCLI")` prefix to match your tool name
- `go.mod` — Update the module path from `github.com/your-org/mycli` to your actual repo
- `cmd/version.go` — The `Version`, `Commit`, and `BuildDate` vars are injected at build time via `-ldflags`
- Add new subcommands in `cmd/` and register them with `rootCmd.AddCommand(...)` in each file's `init()`
