A lightning-fast, fully static developer portfolio built on Next.js App Router with Tailwind CSS — no database, no auth, just MDX-powered blog posts, a filterable project grid, and a contact form wired to a serverless email endpoint.

## Usage

```typescript
// app/blog/[slug]/page.tsx
import { notFound } from 'next/navigation'
import { allPosts } from 'contentlayer/generated'
import { useMDXComponent } from 'next-contentlayer2/hooks'

interface PageProps {
  params: Promise<{ slug: string }>
}

export async function generateStaticParams() {
  return allPosts.map((post) => ({ slug: post.slug }))
}

export async function generateMetadata({ params }: PageProps) {
  const { slug } = await params
  const post = allPosts.find((p) => p.slug === slug)
  if (!post) return {}
  return { title: post.title, description: post.description }
}

export default async function BlogPost({ params }: PageProps) {
  const { slug } = await params
  const post = allPosts.find((p) => p.slug === slug)
  if (!post) notFound()

  const MDXContent = useMDXComponent(post.body.code)

  return (
    <article className="mx-auto max-w-2xl px-4 py-16 prose prose-slate dark:prose-invert">
      <p className="text-sm text-slate-500 not-prose mb-2">
        <time dateTime={post.date}>
          {new Intl.DateTimeFormat('en-US', { dateStyle: 'long' }).format(new Date(post.date))}
        </time>
      </p>
      <h1>{post.title}</h1>
      <MDXContent />
    </article>
  )
}
```

```typescript
// app/contact/route.ts  (serverless email handler)
import { NextRequest, NextResponse } from 'next/server'
import { Resend } from 'resend'

const resend = new Resend(process.env.RESEND_API_KEY)

export async function POST(req: NextRequest) {
  const { name, email, message } = await req.json()

  if (!name || !email || !message) {
    return NextResponse.json({ error: 'Missing fields' }, { status: 400 })
  }

  await resend.emails.send({
    from: 'Portfolio Contact <contact@yourdomain.com>',
    to: 'you@yourdomain.com',
    replyTo: email,
    subject: `New message from ${name}`,
    text: `From: ${name} <${email}>\n\n${message}`,
  })

  return NextResponse.json({ ok: true })
}
```

```typescript
// app/contact/page.tsx
'use client'

import { useState } from 'react'

export default function ContactPage() {
  const [status, setStatus] = useState<'idle' | 'sending' | 'sent' | 'error'>('idle')

  async function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault()
    setStatus('sending')
    const data = Object.fromEntries(new FormData(e.currentTarget))

    const res = await fetch('/contact', { method: 'POST', body: JSON.stringify(data),
      headers: { 'Content-Type': 'application/json' } })

    setStatus(res.ok ? 'sent' : 'error')
  }

  return (
    <main className="mx-auto max-w-lg px-4 py-20">
      <h1 className="text-2xl font-bold mb-6">Get in touch</h1>
      {status === 'sent' ? (
        <p className="text-green-600">Message sent! I'll reply soon.</p>
      ) : (
        <form onSubmit={handleSubmit} className="space-y-4">
          <input required name="name" placeholder="Your name"
            className="w-full rounded-lg border px-4 py-2 text-sm" />
          <input required name="email" type="email" placeholder="your@email.com"
            className="w-full rounded-lg border px-4 py-2 text-sm" />
          <textarea required name="message" rows={5} placeholder="Your message"
            className="w-full rounded-lg border px-4 py-2 text-sm resize-none" />
          <button type="submit" disabled={status === 'sending'}
            className="rounded-lg bg-blue-600 px-6 py-2 text-sm font-medium text-white hover:bg-blue-700 disabled:opacity-50">
            {status === 'sending' ? 'Sending…' : 'Send message'}
          </button>
          {status === 'error' && (
            <p className="text-red-600 text-sm">Something went wrong. Please try again.</p>
          )}
        </form>
      )}
    </main>
  )
}
```

## Notes

All pages except `/contact` (POST handler) are fully static. Run `next build` and `next export` (or deploy to Vercel) and every page renders at build time — zero cold starts. The contact form's POST route remains a serverless function on Vercel. Swap Resend for any transactional email provider (Postmark, SendGrid, Nodemailer) by changing only `app/contact/route.ts`.

## README

### What this includes
- Next.js App Router with `output: 'export'` for fully static HTML output
- MDX blog powered by Contentlayer 2 (`contentlayer2`) with type-safe frontmatter
- `generateStaticParams` for all blog post pages
- Filterable project grid sourced from a typed `data/projects.ts` file
- Contact form with client-side validation wired to a Resend serverless email endpoint
- Tailwind CSS with `@tailwindcss/typography` for blog post prose
- Dark mode via `next-themes`
- Lucide React for icons

### Prerequisites
- Node.js 20.9+
- A [Resend account](https://resend.com) (free tier: 3,000 emails/month) for the contact form

### Create this template

```bash
npx create-next-app@latest my-portfolio --typescript --tailwind --eslint --app --yes
cd my-portfolio
npm install contentlayer2 next-contentlayer2 @mdx-js/loader @mdx-js/react @types/mdx
npm install remark-gfm rehype-highlight
npm install @tailwindcss/typography next-themes lucide-react resend
```

### First things to customize
- `app/page.tsx` — update your name, headline, and bio text
- `data/projects.ts` — replace the sample projects with your actual work
- `content/blog/` — delete the sample post and add your own MDX writing
- `app/contact/route.ts` — replace `to:` and `from:` with your own email addresses
- `app/layout.tsx` — update the `<title>`, `description`, and social links in the footer
- `app/globals.css` — adjust your Tailwind color theme and font choices
